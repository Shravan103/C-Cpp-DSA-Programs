SELECT
    CAST(a.whencLOSED AS DATE) AS report_date,

    /* Section name (can be NULL) */
    s.section AS section_name,

    /* Number of tables */
    COUNT(a.groupid) AS tables_count,

    /* Total guests */
    SUM(a.guests) AS total_guests,

    /* Total amount */
    SUM(a.invoiced) AS total_amount,

    /* Avg Guests / Table */
    CASE
        WHEN COUNT(a.groupid) = 0 THEN 0
        ELSE SUM(a.guests) / COUNT(a.groupid)
    END AS avg_guests_per_table,

    /* Avg Spend / Guest */
    CASE
        WHEN SUM(a.guests) = 0 THEN 0
        ELSE SUM(a.invoiced) / SUM(a.guests)
    END AS avg_spend_per_guest

FROM agroup a

/* Outlet join is required for correct section mapping */
INNER JOIN outlet o
    ON o.outletid = a.outletid

/* Section is optional and derived using table range */
LEFT JOIN section s
    ON s.outletid = a.outletid
   AND a.tablno BETWEEN s.fromtablno AND s.totablno
   AND s.whendeleted IS NULL

WHERE
    a.guests >= 1                  -- TABLES ONLY (not bar tabs)
    AND a.whencLOSED IS NOT NULL   -- only closed tables
    AND a.whencLOSED >= :FromDate
    AND a.whencLOSED <  :ToDate

GROUP BY
    CAST(a.whencLOSED AS DATE),
    s.section

ORDER BY
    CAST(a.whencLOSED AS DATE),
    s.section;
